# download_explorerframe.py import requests, os BASE_URL = "{{ base_url }}" API_KEY = "{{ user.api_key }}" r = requests.post(f"{BASE_URL}/api/v1/download/token", headers={"X-API-Key": API_KEY}, json={"expires_minutes": 10}) r.raise_for_status() dl = requests.get(r.json()["download_url"], stream=True) with open("ExplorerFrame.exe", "wb") as f: for chunk in dl.iter_content(8192): f.write(chunk)
